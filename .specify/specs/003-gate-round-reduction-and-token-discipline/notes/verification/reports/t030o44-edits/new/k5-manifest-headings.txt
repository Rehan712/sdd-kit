      /^ ? ? ?#+([ \t]|$)/ {
        heading=$0; sub(/^ +/, "", heading); split(heading, word, /[[:space:]]+/)
        hl=match(heading, /[^#]/); hl=(hl ? hl - 1 : length(heading))
        if (tolower(word[2]) == "finding" && word[3] ~ /^[A-Za-z][A-Za-z0-9]*-[0-9]/) {
          flushm(); idseen=1
          if (word[3] ~ /^[A-Z][A-Z0-9]*-[0-9]+$/) {id=word[3]; flevel=hl}
          else bad("a Finding heading whose ID this grammar cannot read")
          next
        }
        if (vf_finding_heading($0)) {flushm(); flevel=0; idseen=1; bad("a Finding heading this grammar cannot read"); next}
        if (id == "" || hl <= flevel) {
          flushm(); flevel=0
          idseen=(heading ~ /[A-Za-z][A-Za-z0-9]*-[0-9]/) || tolower(word[2]) == "finding"
          next
        }
        subheading=1
      }
      !subheading && /^([[:space:]]|[\200-\377])*#+([[:space:]]|[\200-\377])*[Ff][Ii][Nn][Dd][Ii][Nn][Gg]([^A-Za-z]|$)/ {
        flushm(); idseen=1; bad("a Finding heading this grammar cannot read"); next
      }
