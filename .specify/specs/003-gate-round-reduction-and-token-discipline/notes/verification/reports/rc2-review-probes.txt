##### capcheck.py
import hashlib,os,sys,glob
spec=sys.argv[1]; cur=sys.argv[2]; ids=sys.argv[3:]
H=lambda b: hashlib.sha256(b).hexdigest()
FILES=["meta.nul","argv.nul","inputs.nul","inputs-after.nul","bindings.nul","output.log","summary.txt"]
KEYS="version id sequence kind start end repo cwd check shell input_before input_after no_tick exit output_digest output_bytes policy key".split()
def meta(p):
    d=open(p,"rb").read().split(b"\0")
    assert d[-1]==b""; d=d[:-1]; assert len(d)%2==0
    m={}
    for i in range(0,len(d),2):
        k=d[i].decode(); assert k in KEYS and k not in m, k; m[k]=d[i+1].decode()
    assert set(m)==set(KEYS); return m
allmeta={}
for led in glob.glob(spec+"/notes/verification/captures/CAP-*"):
    cid=os.path.basename(led); allmeta[cid]=meta(spec+"/notes/captures/"+cid+"/meta.nul")
ok=True
for cid in ids:
    d=spec+"/notes/captures/"+cid
    exp="verification:1\n"+"\n".join(H(open(d+"/"+f,"rb").read())+" "+f for f in FILES)
    comp=open(d+"/COMPLETE").read()
    r=[]
    r.append(("seal", comp.rstrip("\n")==exp))
    r.append(("ledger", open(spec+"/notes/verification/captures/"+cid,"rb").read()==open(d+"/COMPLETE","rb").read()))
    m=meta(d+"/meta.nul")
    r.append(("id/seq", m["id"]==cid and cid.startswith("CAP-"+m["sequence"]+"-") and m["kind"]=="command" and m["version"]=="1"))
    r.append(("exit0", m["exit"]=="0"))
    r.append(("inputs", m["input_before"]==H(open(d+"/inputs.nul","rb").read())==m["input_after"]==H(open(d+"/inputs-after.nul","rb").read())))
    r.append(("current", m["input_before"]==cur))
    log=open(d+"/output.log","rb").read()
    r.append(("output", m["output_digest"]==H(log) and m["output_bytes"]==str(len(log))))
    sh=m["shell"]
    chk=H(open(d+"/argv.nul","rb").read()+b"\0".join(x.encode() for x in [sh,H(open(sh,"rb").read()),m["repo"],m["cwd"],m["policy"]])+b"\0")
    r.append(("check", chk==m["check"]))
    later=[o for o,om in allmeta.items() if int(om["sequence"])>int(m["sequence"]) and om["check"]==m["check"] and om["input_before"]==m["input_before"] and om["exit"]!="0"]
    r.append(("no-later-fail", not later))
    good=all(v for _,v in r); ok&=good
    argv=open(d+"/argv.nul","rb").read().split(b"\0")[:-1]
    print(cid, "OK" if good else "BAD "+str([k for k,v in r if not v]), m["start"], m["end"], m["policy"], "|", (b" ".join(argv[-1:]).decode()[:90]), "| key:", m["key"][:100])
print("ALL-OK" if ok else "SOME-BAD")
